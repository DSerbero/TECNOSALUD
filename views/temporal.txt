import javax.swing.*;
import java.awt.*;
// Asegúrate de tener la librería FlatLaf en tu proyecto
import com.formdev.flatlaf.FlatLightLaf;

public class ClinicaTecnoSaludUI extends JFrame {

    public ClinicaTecnoSaludUI() {
        setTitle("Clínica TECNOSALUD - Sistema de Gestión");
        setSize(900, 600);
        setDefaultCloseOperation(EXIT_ON_CLOSE);
        setLocationRelativeTo(null);

        // Barra de menú moderna
        JMenuBar menuBar = new JMenuBar();

        JMenu menuPacientes = new JMenu("Pacientes");
        menuPacientes.add(new JMenuItem("Registrar Paciente"));
        menuPacientes.add(new JMenuItem("Lista de Pacientes"));

        JMenu menuConsultorios = new JMenu("Consultorios/Habitaciones");
        menuConsultorios.add(new JMenuItem("Asignar Consultorio"));
        menuConsultorios.add(new JMenuItem("Estado de Habitaciones"));

        JMenu menuServicios = new JMenu("Servicios Médicos");
        menuServicios.add(new JMenuItem("Registrar Servicio"));
        menuServicios.add(new JMenuItem("Lista de Servicios"));

        JMenu menuPagos = new JMenu("Pagos");
        menuPagos.add(new JMenuItem("Registrar Pago"));
        menuPagos.add(new JMenuItem("Reporte de Pagos"));

        JMenu menuReportes = new JMenu("Reportes");
        menuReportes.add(new JMenuItem("Pacientes Atendidos"));
        menuReportes.add(new JMenuItem("Montos Recaudados"));
        menuReportes.add(new JMenuItem("Estadísticas por Edad/Género"));

        menuBar.add(menuPacientes);
        menuBar.add(menuConsultorios);
        menuBar.add(menuServicios);
        menuBar.add(menuPagos);
        menuBar.add(menuReportes);

        setJMenuBar(menuBar);

        // Panel central con botones estilizados
        JPanel panelBotones = new JPanel(new GridLayout(3, 2, 15, 15));
        panelBotones.setBorder(BorderFactory.createEmptyBorder(30, 30, 30, 30));

        JButton btnPaciente = new JButton("Registrar Paciente");
        JButton btnConsultorio = new JButton("Asignar Consultorio");
        JButton btnServicio = new JButton("Registrar Servicio");
        JButton btnPago = new JButton("Registrar Pago");
        JButton btnReportes = new JButton("Mostrar Reportes");
        JButton btnSalir = new JButton("Salir");

        // Estilo adicional
        btnSalir.setForeground(Color.RED);
        btnSalir.setFont(new Font("Segoe UI", Font.BOLD, 14));

        panelBotones.add(btnPaciente);
        panelBotones.add(btnConsultorio);
        panelBotones.add(btnServicio);
        panelBotones.add(btnPago);
        panelBotones.add(btnReportes);
        panelBotones.add(btnSalir);

        add(panelBotones, BorderLayout.CENTER);
    }

    public static void main(String[] args) {
        try {
            // Look & Feel moderno
            UIManager.setLookAndFeel(new FlatLightLaf());
        } catch (Exception ex) {
            System.err.println("Error al aplicar FlatLaf: " + ex.getMessage());
        }

        SwingUtilities.invokeLater(() -> {
            new ClinicaTecnoSaludUI().setVisible(true);
        });
    }
}
